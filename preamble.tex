% Shared preamble for every deck in this repo.
% Each deck has its own build root (<deck>/slides.tex) that \input's this file,
% so a deck only ever contains its own content. Nothing here is deck-specific
% except the \title/\author/\institute defaults at the bottom, which a deck's
% slides.tex may override before \begin{document}.
\documentclass[aspectratio=169, dvipsnames, theorems=ams, t]{beamer}
\usetheme{princeton}

\usepackage[T1]{fontenc}
%\usepackage[final]{microtype}
% NOTE: text font (Carlito, a Calibri clone) is set by the princeton theme.
% Keep eulervm for math only.
\usepackage[euler-digits,euler-hat-accent]{eulervm}

\usepackage{amsmath,amssymb,amsfonts,mathtools,bm}
\usepackage{graphicx}
% Builds run with the working directory set to the deck folder, so shared assets
% at the repo root (logos/, ...) sit one or two levels up depending on how deep
% the deck lives. Deck-local paths (figures/, tables/) still resolve first.
\graphicspath{{./}{../}{../../}}
\usepackage{caption}
\usepackage{subcaption}
\usepackage{booktabs,arydshln}
\usepackage[ruled,vlined]{algorithm2e}
\usepackage{algorithmic}
\setlength{\marginparwidth}{2cm} % required by todonotes to avoid layout warnings
\usepackage{todonotes}
\usepackage{float}
\usepackage{multirow}
\usepackage{textcomp}
\usepackage{gensymb}
\usepackage{wasysym}
\usepackage{ifsym}
\usepackage{relsize}
\usepackage{tabto}
\usepackage{filecontents}
\usepackage{framed}
\usepackage{tcolorbox}
\usepackage{array}
\usepackage{xcolor}
\usepackage{pgfplots}
\usepackage{tikz}
\usepackage{enumitem}
\setitemize{label=\usebeamerfont*{itemize item}%
  \usebeamercolor[fg]{itemize item}
  \usebeamertemplate{itemize item}}
\usepackage{ulem}
\usepackage{pdfpages}

% Citations: keep using \cite{...} but render as "Author et al. [1]"
% This uses natbib in numeric mode; \cite is aliased to \citet.
\usepackage[authoryear,square]{natbib}
% Prevent natbib from inserting a \section*{References} (would trigger \AtBeginSection)
\renewcommand{\bibsection}{}

\setbeamertemplate{theorems}[numbered]

\pgfplotsset{compat=1.15}
\usepgfplotslibrary{fillbetween}
\usetikzlibrary{calc}
\usetikzlibrary{colorbrewer}
\usetikzlibrary{shapes}
\usetikzlibrary{overlay-beamer-styles}
\usepgfplotslibrary{colorbrewer}
% Libraries loaded here, not inside a figure file: \usetikzlibrary within a
% frame defines its keys locally, so a second frame reusing the library would
% not see them. The slides-figure skill assumes all of these are available.
\usetikzlibrary{positioning}
\usetikzlibrary{arrows.meta}
\usetikzlibrary{patterns}
\usetikzlibrary{fit}
\usetikzlibrary{backgrounds}
\usepgfplotslibrary{groupplots}

% Color-blind-safe categorical palette (Okabe-Ito) for figures, available as
% c0..c5 to both the deck and the slides-figure preview. Use `princeton` for the
% series you want to emphasize; avoid pairing it with c3, which is close to it.
\definecolor{c0}{HTML}{0072B2} % blue
\definecolor{c1}{HTML}{E69F00} % amber
\definecolor{c2}{HTML}{009E73} % green
\definecolor{c3}{HTML}{D55E00} % vermillion
\definecolor{c4}{HTML}{CC79A7} % purple
\definecolor{c5}{HTML}{56B4E9} % sky blue

\newcommand{\mainblock}[2] {
\begin{tcolorbox}[colback=princeton!5,colframe=princeton!90!black,title={\centering #1}]
#2
\end{tcolorbox}
}

\newcommand{\boxedblock}[1] {
\begin{tcolorbox}[colback=princeton!5,colframe=princeton!75!black]
\centering
#1
\end{tcolorbox}
}

% Defaults; override in a deck's slides.tex after \input{preamble.tex}.
\title[\today]{Meeting}
\subtitle{}
\date{\today}

\author{Anderson Lee}
